// The development overlay captures the ordered ports and vertices after both
// initial visibility stages. It does not perform or replace any Go layout work.
type pilotOVGOwner struct {
    Node int `json:"node"`
    Sides uint8 `json:"sides"`
    Center bool `json:"center"`
}

type pilotOVGVertex struct {
    Point struct {
        X float64 `json:"x"`
        Y float64 `json:"y"`
    } `json:"point"`
    Owners []pilotOVGOwner `json:"owners"`
}

type pilotOVGRectangle struct {
    ID string `json:"id"`
    X float64 `json:"x"`
    Y float64 `json:"y"`
    Width float64 `json:"width"`
    Height float64 `json:"height"`
}

func tracePilotOVGSeed(ovg *OVG, graph *layoutgraph.Graph, ordinary bool, stage string) {
    if os.Getenv("TALA_PILOT_TRACE_OVG_SEED") != "1" {
        return
    }
    if !ordinary || len(ovg.NodesInsideBoundingBox) != len(graph.Nodes) {
        return
    }
    linked := stage == "LINKS"
    if (linked || stage == "TUNNELS") && (len(graph.NodeToTree) != 0 || len(graph.Clusters) != 0) {
        return
    }
    nodes := make([]pilotOVGRectangle, 0, len(graph.Nodes))
    slots := make([][]int, len(graph.Nodes))
    vertices := make([]pilotOVGVertex, 0, len(ovg.Nodes))
    indices := make(map[*OVGNode]int, len(ovg.Nodes))
    sides := [...]geo.Orientation{geo.Top, geo.Left, geo.Bottom, geo.Right}
    for i, vertex := range ovg.Nodes {
        indices[vertex] = i
        entry := pilotOVGVertex{Owners: make([]pilotOVGOwner, 0, len(vertex.portOwners()))}
        entry.Point.X, entry.Point.Y = vertex.X, vertex.Y
        for j, node := range graph.Nodes {
            directions, ok := vertex.portDirectionsFor(node)
            if !ok {
                continue
            }
            var flags uint8
            for side, direction := range sides {
                if directions.has(direction) {
                    flags |= 1 << side
                }
            }
            entry.Owners = append(entry.Owners, pilotOVGOwner{j, flags, vertex.isCenterPortOf(node)})
        }
        if len(entry.Owners) != len(vertex.portOwners()) {
            return
        }
        vertices = append(vertices, entry)
    }
    for i, node := range graph.Nodes {
        ports := ovg.Ports[node]
        if node.Shape.GetType() != "Square" || node.IsContainer() || node.IsInvisible ||
            node.Cluster != nil || node.Sequence != nil || node.FixedTopLeft != nil || (!linked && stage != "TUNNELS" && len(ports) != 12) {
            return
        }
        slots[i] = make([]int, 0, len(ports))
        for _, port := range ports {
            index, ok := indices[port]
            if !ok {
                return
            }
            slots[i] = append(slots[i], index)
        }
        nodes = append(nodes, pilotOVGRectangle{node.DebugID(), node.TopLeft.X, node.TopLeft.Y, node.Width, node.Height})
    }
    result := struct {
        Nodes []pilotOVGRectangle `json:"nodes"`
        Edges *[][2]int `json:"edges,omitempty"`
        Expected struct {
            Vertices []pilotOVGVertex `json:"vertices"`
            PortsByNode [][]int `json:"portsByNode"`
            TunnelLinks *[][2]int `json:"tunnelLinks,omitempty"`
            Links *[][2]int `json:"links,omitempty"`
            CentersByNode *[]int `json:"centersByNode,omitempty"`
            TunnelLinkCount *int `json:"tunnelLinkCount,omitempty"`
            NearPortOwners *[][]int `json:"nearPortOwners,omitempty"`
        } `json:"expected"`
    }{Nodes: nodes}
    result.Expected.Vertices = vertices
    result.Expected.PortsByNode = slots
    if stage == "TUNNELS" {
        links := make([][2]int, 0, len(ovg.Edges))
        for _, edge := range ovg.Edges {
            from, hasFrom := indices[edge.From]
            to, hasTo := indices[edge.To]
            if !hasFrom || !hasTo || !edge.From.IsTunnel || !edge.To.IsTunnel {
                return
            }
            links = append(links, [2]int{from, to})
        }
        result.Expected.TunnelLinks = &links
    }
    if linked {
        links := make([][2]int, 0, len(ovg.Edges))
        tunnelCount := 0
        for _, edge := range ovg.Edges {
            from, hasFrom := indices[edge.From]
            to, hasTo := indices[edge.To]
            if !hasFrom || !hasTo {
                return
            }
            if edge.From.IsTunnel && edge.To.IsTunnel {
                tunnelCount++
            }
            links = append(links, [2]int{from, to})
        }
        centers := make([]int, 0, len(graph.Nodes))
        for _, node := range graph.Nodes {
            index, ok := indices[ovg.Centers[node]]
            if !ok {
                return
            }
            centers = append(centers, index)
        }
        near := make([][]int, len(ovg.Nodes))
        for i, vertex := range ovg.Nodes {
            near[i] = make([]int, 0, len(vertex.IsNearPort))
            for j, owner := range graph.Nodes {
                if _, ok := vertex.IsNearPort[owner]; ok {
                    near[i] = append(near[i], j)
                }
            }
        }
        result.Expected.Links = &links
        result.Expected.CentersByNode = &centers
        result.Expected.TunnelLinkCount = &tunnelCount
        result.Expected.NearPortOwners = &near
    }
    prefix := "TALA_PILOT_OVG_" + stage
    if stage != "SEED" {
        graphIndices := make(map[*layoutgraph.Node]int, len(graph.Nodes))
        for i, node := range graph.Nodes {
            graphIndices[node] = i
        }
        edges := make([][2]int, 0, len(graph.Edges))
        for _, edge := range graph.Edges {
            from, hasFrom := graphIndices[edge.From]
            to, hasTo := graphIndices[edge.To]
            if !hasFrom || !hasTo {
                return
            }
            edges = append(edges, [2]int{from, to})
        }
        result.Edges = &edges
    }
    encoded, err := json.Marshal(result)
    if err != nil {
        panic(err)
    }
    fmt.Fprintf(os.Stderr, "%s %s\n", prefix, encoded)
}
